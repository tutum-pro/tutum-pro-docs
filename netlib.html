library_books tutum-netlib

Go Client Library for Certificate Retrieval - Shared Object (.so) with JDBC-like API

info Overview

tutum-netlib is a Go-based shared library that provides a simple, JDBC-like API for retrieving X.509 certificates from the Tutum Engine. It serves as the communication layer between CSI drivers (both Kubernetes and Docker) and the tutum-engine backend.

lightbulb
Key Design: The library is compiled as a shared object (.so) file, making it linkable from both Go and C applications. This allows both CSI drivers to share the same certificate retrieval logic.

Certificate Retrieval

warning Breaking Change (v1.5.0): GetCertificate now returns []*Certificate (slice) instead of singular *Certificate to support semantic consistency. 
// GetCertificate retrieves certificates by name (returns 0-2 certificates)
// Semantic consistency: X509 and secret with same name/group/namespace
// are returned together when both exist and have is_effective=true
func (c *Client) GetCertificate(ctx context.Context, name, groupID, namespace string) ([]*Certificate, error)

// GetCertificateVersionWithAuth retrieves certificates with version, channel, and auth token
func (c *Client) GetCertificateVersionWithAuth(ctx context.Context, name, groupID, namespace string,
    version int, workload *WorkloadContext, channel, authToken string) ([]*Certificate, error)

// WorkloadContext provides workload info for audit logging
type WorkloadContext struct {
    Type      string  // k8s-pod, k8s-deployment, docker-container
    Name      string  // Name of the workload
    Namespace string  // Workload's namespace
    UID       string  // Unique workload identifier
}

// Certificate contains the retrieved certificate data
type Certificate struct {
    ID             string    // Certificate UUID
    Name           string    // Certificate name
    GroupID        string    // Group UUID
    GroupName      string    // Group name
    Namespace      string    // Namespace
    Version        int32     // Certificate version
    Type           string    // "X509" or "secret"
    CertificatePEM string    // PEM-encoded certificate (or secret data)
    PrivateKeyPEM  string    // PEM-encoded private key (empty for secrets)
    CAChainPEM     string    // PEM-encoded CA chain
    IsEffective    bool      // Whether this is the effective version
    EffectiveSince time.Time // When version became effective
    CreatedAt      time.Time // Creation timestamp
    UpdatedAt      time.Time // Last update timestamp
}

code Usage Examples

Basic Usage (Go)

package main

import (
    "context"
    "log"
    "github.com/tutum/netlib"
)

func main() {
    // Connect to tutum-engine
    client, err := netlib.Connect("localhost:9090", netlib.WithInsecure())
    if err != nil {
        log.Fatal(err)
    }
    defer client.Close()

    // Retrieve certificate
    cert, err := client.GetCertificate(context.Background(), &netlib.GetCertificateRequest{
        Name:      "my-app-cert",
        GroupID:   "group-uuid",
        Namespace: "production",
    })
    if err != nil {
        log.Fatal(err)
    }

    log.Printf("Certificate: %s", cert.CertificatePEM)
    log.Printf("Valid until: %s", cert.NotAfter)
}

With Authentication

With mTLS

client, err := netlib.Connect(
    "tutum-engine.tutum-system:9090",
    netlib.WithTLS(
        "/etc/tutum/client.crt",
        "/etc/tutum/client.key",
        "/etc/tutum/ca.crt",
    ),
    netlib.WithTimeout(10 * time.Second),
)

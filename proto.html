code Protocol Buffers (tutum-proto)

Shared Protocol Buffer Definitions for gRPC API Contract

info Overview

tutum-proto contains the Protocol Buffer v3 definitions that define the gRPC API contract between tutum-engine and its clients (tutum-netlib, CSI drivers, and external applications).

Contract Definition

Single source of truth for API schema

api Service Definitions

CertificateService

syntax = "proto3";

package tutum.v1;

service CertificateService {
  // GetCertificate retrieves a certificate by name
  rpc GetCertificate(GetCertificateRequest) returns (GetCertificateResponse);

  // ListCertificates returns a paginated list of certificates
  rpc ListCertificates(ListCertificatesRequest) returns (ListCertificatesResponse);

  // WatchCertificate streams certificate updates
  rpc WatchCertificate(WatchCertificateRequest) returns (stream WatchCertificateResponse);

  // GetCertificateMetadata returns certificate metadata without sensitive data
  rpc GetCertificateMetadata(GetCertificateMetadataRequest) returns (CertificateMetadata);
}

GroupService

service GroupService {
  // CreateGroup creates a new certificate group
  rpc CreateGroup(CreateGroupRequest) returns (CreateGroupResponse);

  // GetGroup retrieves a group by ID
  rpc GetGroup(GetGroupRequest) returns (GetGroupResponse);

  // ListGroups returns all groups
  rpc ListGroups(ListGroupsRequest) returns (ListGroupsResponse);

  // UpdateGroup updates a group
  rpc UpdateGroup(UpdateGroupRequest) returns (UpdateGroupResponse);

  // DeleteGroup deletes a group
  rpc DeleteGroup(DeleteGroupRequest) returns (DeleteGroupResponse);
}

data_object Message Types

GetCertificateRequest

message GetCertificateRequest {
  string name = 1;                    // Certificate name (required)
  string group_id = 2;                // Group UUID (required)
  string namespace = 3;               // Namespace name (required)
  int32 version = 4;                  // Specific version (0 = latest)
  string channel = 5;                 // Distribution channel name
  string auth_token = 6;              // Authentication token
  string workload_type = 7;           // k8s-pod, k8s-deployment, etc.
  string workload_name = 8;           // Workload name for audit
  string workload_namespace = 9;      // Workload namespace for audit
}

GetCertificateResponse

warning Breaking Change (v1.5.0): Response changed from singular certificate to repeated certificates for semantic consistency support. 
message GetCertificateResponse {
  // Returns 0-2 certificates with is_effective=true
  // Semantic consistency: X509 and secret with same name/group/namespace
  // are returned together in a single response
  repeated Certificate certificates = 1;
}

message Certificate {
  string id = 1;                      // Certificate UUID
  string name = 2;                    // Certificate name
  string group_id = 3;                // Group UUID
  string group_name = 4;              // Group name
  string namespace = 5;               // Namespace
  int32 version = 6;                  // Version number
  CertificateType type = 7;           // X509 or SECRET
  CertificateStatus status = 8;       // Current status
  string certificate_pem = 9;         // PEM-encoded certificate (or secret data)
  string private_key_pem = 10;        // PEM-encoded private key (empty for secrets)
  string ca_chain_pem = 11;           // PEM-encoded CA chain
  X509Metadata x509_metadata = 12;    // X.509 metadata (null for secrets)
  google.protobuf.Timestamp created_at = 13;
  google.protobuf.Timestamp updated_at = 14;
  map custom_metadata = 15;
  bool is_effective = 16;             // Whether this is the effective version
  google.protobuf.Timestamp effective_since = 17; // When version became effective
}

X509Metadata

message X509Metadata {
  string subject = 1;                 // Certificate subject (CN=...)
  string issuer = 2;                  // Certificate issuer
  string serial_number = 3;           // Serial number (hex)
  google.protobuf.Timestamp not_before = 4;
  google.protobuf.Timestamp not_after = 5;
  string fingerprint_sha256 = 6;      // SHA-256 fingerprint
  repeated string subject_alt_names = 7;
  repeated string key_usage = 8;
  repeated string extended_key_usage = 9;
  bool is_ca = 10;
}

Using Buf

# Install buf
brew install bufbuild/buf/buf

# Generate code for all languages
buf generate

# Lint proto files
buf lint

# Check for breaking changes
buf breaking --against '.git#branch=main'

buf.gen.yaml Configuration

version: v1
plugins:
  # Go code generation
  - plugin: go
    out: gen/go
    opt:
      - paths=source_relative
  - plugin: go-grpc
    out: gen/go
    opt:
      - paths=source_relative

  # Python code generation
  - plugin: python
    out: gen/python
  - plugin: grpc_python
    out: gen/python

  # Java code generation
  - plugin: java
    out: gen/java

Using Generated Code (Go)

import (
    tutumv1 "github.com/tutum/proto/gen/go/tutum/v1"
    "google.golang.org/grpc"
)

// Create client
conn, _ := grpc.Dial("localhost:9090", grpc.WithInsecure())
client := tutumv1.NewCertificateServiceClient(conn)

// Make request
resp, err := client.GetCertificate(ctx, &tutumv1.GetCertificateRequest{
    Name:      "my-cert",
    GroupId:   "group-uuid",
    Namespace: "default",
})

// Access response
fmt.Println(resp.Certificate.CertificatePem)
fmt.Println(resp.Certificate.X509Metadata.Subject)

info
Backward Compatibility: Breaking changes are only introduced in new major versions. Fields marked as deprecated will be supported for at least 2 minor versions before removal.
